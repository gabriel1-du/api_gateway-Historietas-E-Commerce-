# El gateway usa .env (no .env.prod). Ya existe en el repo; si falta, crealo:
#   SERVER_PORT=8888

nano .env

bash -n deploy.sh
./deploy.sh --check            # solo valida, no modifica nada
./deploy.sh --dry-run          # muestra lo que haría
./deploy.sh feature/StartUp-Info-Request-Logs   # despliega de verdad (o master)

# Comandos útiles después del deploy:
sudo docker compose -f docker-compose.prod.yml --env-file .env logs -f apigateway

sudo docker compose -f docker-compose.prod.yml --env-file .env restart apigateway
